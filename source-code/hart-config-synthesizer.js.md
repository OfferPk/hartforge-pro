 \* HARTForge Pro \- HART-IP Industrial Firewall & IDS Configuration Synthesizer

class HARTConfigSynthesizer {  
  generateNftables(config \= {}) {  
    const gwIp \= config.gwIp \|\| '192.168.10.20';  
    const scadaIp \= config.scadaIp \|\| '192.168.10.100';  
    const amsIp \= config.amsIp \|\| '192.168.10.50';

    return \`\#\!/usr/sbin/nft \-f  
\# HARTForge Pro \- Linux nftables Industrial Process Conduit Firewall  
\# Hardened Ruleset for HART-IP (UDP/TCP Port 5094\) Gateway Protection  
\# Reference: IEC 62443-3-3 Zone & Conduit Security

flush ruleset

table inet hart\_firewall {  
    \# Authorized Distributed Control Systems (DCS) / SCADA Hosts  
    set authorized\_dcs {  
        type ipv4\_addr  
        elements \= { \${scadaIp} }  
    }

    \# Asset Management Stations (AMS / Field Device Tools)  
    set asset\_management {  
        type ipv4\_addr  
        elements \= { \${amsIp} }  
    }

    \# Protected WirelessHART / HART-IP Gateways  
    set hart\_gateways {  
        type ipv4\_addr  
        elements \= { \${gwIp} }  
    }

    chain input {  
        type filter hook input priority filter; policy drop;

        iif "lo" accept  
        ct state established,related accept  
        ct state invalid drop

        ip protocol icmp limit rate 5/second accept  
    }

    chain forward {  
        type filter hook forward priority filter; policy drop;

        ct state established,related accept  
        ct state invalid drop

        \# 1\. Allow HART-IP Telemetry (UDP Port 5094\) from Gateways to DCS / SCADA  
        ip saddr @hart\_gateways ip daddr @authorized\_dcs udp sport 5094 accept  
        ip saddr @authorized\_dcs ip daddr @hart\_gateways udp dport 5094 accept

        \# 2\. Allow HART-IP Diagnostics (TCP Port 5094\) strictly from Asset Management Station  
        ip saddr @asset\_management ip daddr @hart\_gateways tcp dport 5094 ct state new limit rate 15/minute accept  
        ip saddr @asset\_management ip daddr @hart\_gateways tcp dport 5094 accept

        \# Log unauthorized attempts to access HART-IP process ports  
        ip daddr @hart\_gateways udp dport 5094 log prefix "\[HART-IP-UNAUTH-UDP\] " drop  
        ip daddr @hart\_gateways tcp dport 5094 log prefix "\[HART-IP-UNAUTH-TCP\] " drop  
    }

    chain output {  
        type filter hook output priority filter; policy accept;  
    }  
}  
\`;  
  }

  generateSuricataRules(config \= {}) {  
    const sidBase \= config.sidBase \|\| 20261690;  
    return \`\# Suricata L7 Industrial IDS Rules for HART-IP Protocol (IEC 62591\)  
\# Generated by HARTForge Pro \- v1.0.0  
\# Target: HART-IP (UDP/TCP Port 5094\) Process Field Transmitters & Gateways

\# 1\. Detect HART-IP Command 35 (Write Primary Variable Range Values / Re-span)  
alert udp any any \-\> any 5094 (msg:"HART-IP Write Range Values Command (Cmd 35\) Detected"; content:"\|01\|"; offset:0; depth:1; content:"\|03\|"; offset:2; depth:1; content:"\|23\|"; distance:6; within:4; classtype:attempted-admin; sid:\${sidBase \+ 1}; rev:1;)

\# 2\. Detect HART-IP Command 45 (Trim DAC Current / Calibration Zero-Gain Adjust)  
alert udp any any \-\> any 5094 (msg:"HART-IP CRITICAL \- Trim DAC Analog Current (Cmd 45\) Injected"; content:"\|01\|"; offset:0; depth:1; content:"\|03\|"; offset:2; depth:1; content:"\|2d\|"; distance:6; within:4; priority:1; classtype:attempted-admin; sid:\${sidBase \+ 2}; rev:1;)

\# 3\. Detect HART-IP Command 42 (Master Reset / Device Reboot Command)  
alert udp any any \-\> any 5094 (msg:"HART-IP Field Device Master Reset (Cmd 42\) Injected"; content:"\|01\|"; offset:0; depth:1; content:"\|03\|"; offset:2; depth:1; content:"\|2a\|"; distance:6; within:4; classtype:successful-dos; sid:\${sidBase \+ 3}; rev:1;)

\# 4\. Detect NAMUR NE107 Field Device Malfunction Alert in Response  
alert udp any 5094 \-\> any any (msg:"HART-IP NAMUR NE107 Field Device Hardware Malfunction Flagged"; content:"\|01 01 03\|"; offset:0; depth:3; byte\_test:1,&,0x80,15; classtype:system-reconfiguration; sid:\${sidBase \+ 4}; rev:1;)

\# 5\. Detect HART-IP Session Initiation Attempt  
alert tcp any any \-\> any 5094 (msg:"HART-IP Management Session Initiate (MsgId 0x00)"; content:"\|01 00 00\|"; offset:0; depth:3; classtype:protocol-command-decode; sid:\${sidBase \+ 5}; rev:1;)  
\`;  
  }  
}
